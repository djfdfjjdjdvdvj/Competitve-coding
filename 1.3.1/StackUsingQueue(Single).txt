class Queue:
    def __init__(self):
        self.s = []

    def enqueue(self, x):
        self.s.append(x)

    def dequeue(self):
        if not self.s:
            print("Queue Underflow")
            return

        x = self.s.pop()

        if not self.s:
            print("Dequeued:", x)
            return

        self.dequeue()

        self.s.append(x)

    def front(self):
        if not self.s:
            print("Queue is Empty")
            return

        x = self.s.pop()

        if not self.s:
            print("Front:", x)
            self.s.append(x)
            return

        self.front()

        self.s.append(x)


q = Queue()

q.enqueue(10)
q.enqueue(20)
q.enqueue(30)

q.front()
q.dequeue()
