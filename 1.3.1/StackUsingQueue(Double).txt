class Queue:
    def __init__(self):
        self.s1 = []
        self.s2 = []

    def enqueue(self, x):
        self.s1.append(x)

    def dequeue(self):
        if not self.s1 and not self.s2:
            print("Queue Underflow")
            return

        if not self.s2:
            while self.s1:
                self.s2.append(self.s1.pop())

        print("Dequeued:", self.s2.pop())

    def front(self):
        if not self.s1 and not self.s2:
            print("Queue is Empty")
            return

        if not self.s2:
            while self.s1:
                self.s2.append(self.s1.pop())

        print("Front:", self.s2[-1])


q = Queue()

q.enqueue(10)
q.enqueue(20)
q.enqueue(30)

q.front()
q.dequeue()
